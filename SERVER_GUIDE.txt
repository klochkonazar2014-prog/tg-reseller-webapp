import os
content = open('/home/octorent/auto_buyer.py', 'r').read()
sync_func = """
async def sync_rented_tc_links():
    \"\"\"Фоновая задача для автоматической привязки tc_link к Fragment (дожим)\"\"\"
    import logging, aiohttp, database as db, asyncio
    logging.info("🔗 Воркер синхронизации TonConnect запущен...")
    while True:
        try:
            async with db.aiosqlite.connect(db.DB_PATH) as conn:
                conn.row_factory = db.aiosqlite.Row
                async with conn.execute(
                    "SELECT * FROM orders WHERE status IN ('rented', 'active') AND tc_link IS NOT NULL AND api_token IS NOT NULL"
                ) as cursor:
                    orders = await cursor.fetchall()
            if orders:
                async with aiohttp.ClientSession() as session:
                    for order in orders:
                        try:
                            logging.info(f"🔄 [Sync] Пробую привязку Fragment для заказа #{order['id']}...")
                            url_tc = f"https://api.marketapp.ws/v1/rent/{order['nft_address']}/tonconnect/"
                            payload_tc = {"tonconnect_url": order['tc_link']}
                            headers = {"Authorization": order['api_token'], "Content-Type": "application/json"}
                            async with session.post(url_tc, headers=headers, json=payload_tc, timeout=15) as resp:
                                body = await resp.text()
                                logging.info(f"📥 [Sync] Ответ #{order['id']}: {resp.status} {body}")
                        except Exception as e_order:
                            logging.error(f"❌ [Sync] Ошибка #{order['id']}: {e_order}")
        except Exception as e:
            logging.error(f"❌ [Sync] Ошибка воркера: {e}")
        await asyncio.sleep(120)
"""
if "async def sync_rented_tc_links():" not in content:
    content = content.replace("async def main():", sync_func + "\nasync def main():")
if "sync_rented_tc_links()" not in content:
    content = content.replace("process_referral_withdrawals()", "process_referral_withdrawals(),\n        sync_rented_tc_links()")
open('/home/octorent/auto_buyer.py', 'w').write(content)
